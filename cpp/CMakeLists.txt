cmake_minimum_required(VERSION 3.15)
project(wakeup_axera_sdk LANGUAGES CXX C)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include_directories(include)
add_library(wakeup_axera_sdk STATIC src/model_runner.cpp)
target_sources(wakeup_axera_sdk PRIVATE src/wake_word.cpp)
target_include_directories(wakeup_axera_sdk PUBLIC include)
add_executable(model_example examples/main.cpp)
if(AX_RUNTIME_ROOT)
  target_include_directories(wakeup_axera_sdk PUBLIC ${AX_RUNTIME_ROOT}/include)
  target_link_directories(wakeup_axera_sdk PUBLIC ${AX_RUNTIME_ROOT}/lib)
  target_link_libraries(wakeup_axera_sdk PUBLIC ax_engine ax_sys ax_interpreter)
endif()
target_link_libraries(model_example PRIVATE wakeup_axera_sdk)
